<head>

  <style>
    html, body { margin: 0; width: 100%; height: 100%; overflow: hidden; }

    }
    .label small { display: block; font-size: 13px; letter-spacing: .12em; }
    .label strong { display: block; margin-top: 3px; font-size: 38px; font-weight: 700; }
    .label span { display: block; margin-top: 3px; font-size: 14px; }
    .progress {

<body>
  <div class="label" aria-live="polite">
    <small>사계절의 나무</small>
    <strong id="season">봄</strong>
    <span id="description">꽃이 피어납니다</span>
    <div class="progress"><i id="progress"></i></div>
  </div>

  <script type="module">
    import * as THREE from "https://cdn.jsdelivr.net/npm/three@0.160.1/build/three.module.js";

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
    renderer.setSize(innerWidth, innerHeight);
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    renderer.outputColorSpace = THREE.SRGBColorSpace;
    document.body.appendChild(renderer.domElement);

    const scene = new THREE.Scene();
    scene.fog = new THREE.Fog("#cbdfea", 13, 34);

    const camera = new THREE.PerspectiveCamera(45, innerWidth / innerHeight, 0.1, 150);
    camera.position.set(7, 4.5, 10);
    camera.lookAt(0, 2, 0);

    const skyMaterial = new THREE.ShaderMaterial({
      side: THREE.BackSide,
      depthWrite: false,
      uniforms: {
        topColor: { value: new THREE.Color() },
        horizonColor: { value: new THREE.Color() }
      },
      vertexShader: `
        varying vec3 vPosition;
        void main() {
          vPosition = position;
          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
        }
      `,
      fragmentShader: `
        varying vec3 vPosition;
        uniform vec3 topColor;
        uniform vec3 horizonColor;
        void main() {
          float h = normalize(vPosition).y;
          float t = smoothstep(-0.25, 0.8, h);
          gl_FragColor = vec4(mix(horizonColor, topColor, t), 1.0);
        }
      `
    });
    scene.add(new THREE.Mesh(new THREE.SphereGeometry(90, 32, 16), skyMaterial));

    const ambient = new THREE.HemisphereLight(0xffffff, 0x647061, 2.1);
    scene.add(ambient);

    const sunLight = new THREE.DirectionalLight(0xffffff, 2.2);
    sunLight.position.set(-5, 10, 5);
    sunLight.castShadow = true;
    sunLight.shadow.mapSize.set(2048, 2048);
    sunLight.shadow.camera.left = -9;
    sunLight.shadow.camera.right = 9;
    sunLight.shadow.camera.top = 9;
    sunLight.shadow.camera.bottom = -9;
    scene.add(sunLight);

    const sun = new THREE.Mesh(
      new THREE.SphereGeometry(1.05, 24, 16),
      new THREE.MeshBasicMaterial({ color: 0xfff1c6, fog: false })
    );
    sun.position.set(-10, 11, -22);
    scene.add(sun);

    const hillMaterial = new THREE.MeshStandardMaterial({ color: 0x8fbd78, roughness: 1 });
    const plainMaterial = new THREE.MeshStandardMaterial({ color: 0x9ec687, roughness: 1 });
    const plain = new THREE.Mesh(new THREE.PlaneGeometry(100, 100), plainMaterial);
    plain.rotation.x = -Math.PI / 2;
    plain.position.y = -0.53;
    plain.receiveShadow = true;
    scene.add(plain);

    const hill = new THREE.Mesh(new THREE.SphereGeometry(1, 64, 32), hillMaterial);
    hill.scale.set(5, 1.2, 4.2);
    hill.position.y = -0.45;
    hill.receiveShadow = true;
    scene.add(hill);

    let seed = 1729;
    function random() {
      seed = (seed * 1664525 + 1013904223) >>> 0;
      return seed / 4294967296;
    }
    const range = (a, b) => a + (b - a) * random();

    const tree = new THREE.Group();
    tree.position.y = 0.73;
    scene.add(tree);
    const bark = new THREE.MeshStandardMaterial({ color: 0x71523e, roughness: 1 });

    function addBranch(a, b, bottomRadius, topRadius) {
      const direction = new THREE.Vector3().subVectors(b, a);
      const mesh = new THREE.Mesh(
        new THREE.CylinderGeometry(topRadius, bottomRadius, direction.length(), 7),
        bark
      );
      mesh.position.copy(a).add(b).multiplyScalar(0.5);
      mesh.quaternion.setFromUnitVectors(new THREE.Vector3(0, 1, 0), direction.normalize());
      mesh.castShadow = true;
      tree.add(mesh);
    }

    const trunk = [
      new THREE.Vector3(0, 0, 0),
      new THREE.Vector3(0.06, 0.95, 0.02),
      new THREE.Vector3(-0.04, 1.85, -0.03),
      new THREE.Vector3(0.07, 2.7, 0.04),
      new THREE.Vector3(0, 3.38, 0)
    ];
    [0.31, 0.25, 0.17, 0.09].forEach((r, i) =>
      addBranch(trunk[i], trunk[i + 1], r, r * 0.72)
    );

    for (let layer = 0; layer < 4; layer++) {
      const count = layer === 0 ? 6 : 7;
      for (let i = 0; i < count; i++) {
        const angle = i * Math.PI * 2 / count + layer * 0.53;
        const start = new THREE.Vector3(0, 1.25 + layer * 0.43, 0);
        const length = 1.05 + layer * 0.13;
        const end = new THREE.Vector3(
          Math.cos(angle) * length,
          start.y + 0.52 + layer * 0.04,
          Math.sin(angle) * length
        );
        addBranch(start, end, 0.12 - layer * 0.014, 0.055);
        for (const side of [-1, 1]) {
          const tipAngle = angle + side * 0.42;
          const tip = new THREE.Vector3(
            end.x + Math.cos(tipAngle) * 0.66,
            end.y + range(0.35, 0.7),
            end.z + Math.sin(tipAngle) * 0.66
          );
          addBranch(end, tip, 0.055, 0.015);
        }
      }
    }

    const leafGeometry = new THREE.SphereGeometry(1, 6, 4);
    const flowerVertices = [];
    for (let p = 0; p < 5; p++) {
      const a = p * Math.PI * 2 / 5;
      const radial = (r, offset = 0) => [
        Math.cos(a + offset) * r,
        Math.sin(a + offset) * r,
        0
      ];
      const center = radial(0.035);
      const left = radial(0.14, -0.42);
      const tip = radial(0.23);
      const right = radial(0.14, 0.42);
      flowerVertices.push(...center, ...left, ...tip, ...center, ...tip, ...right);
    }
    const flowerGeometry = new THREE.BufferGeometry();
    flowerGeometry.setAttribute(
      "position",
      new THREE.Float32BufferAttribute(flowerVertices, 3)
    );
    flowerGeometry.computeVertexNormals();

    const count = 760;
    const positions = [];
    for (let i = 0; i < count; i++) {
      let x, y, z;
      do {
        x = range(-1, 1);
        y = range(-1, 1);
        z = range(-1, 1);
      } while (x * x + y * y + z * z > 1);
      positions.push(new THREE.Vector3(x * 1.93, 2.83 + y * 1.3, z * 1.62));
    }

    function makeCanopy(geometry, palette, flower = false) {
      const material = flower
        ? new THREE.MeshBasicMaterial({
            color: 0xffffff, side: THREE.DoubleSide,
            transparent: true, opacity: 1, depthWrite: false
          })
        : new THREE.MeshStandardMaterial({
            color: 0xffffff, roughness: 0.9,
            transparent: true, opacity: 1, depthWrite: false
          });
      const mesh = new THREE.InstancedMesh(geometry, material, count);
      const dummy = new THREE.Object3D();
      for (let i = 0; i < count; i++) {
        dummy.position.copy(positions[i]);
        dummy.rotation.set(range(0, Math.PI), range(0, Math.PI * 2), range(0, Math.PI));
        const size = flower ? range(0.65, 1.08) : range(0.1, 0.2);
        dummy.scale.set(
          flower ? size : size * 0.72,
          flower ? size : size * 1.55,
          flower ? size : size * 0.4
        );
        dummy.updateMatrix();
        mesh.setMatrixAt(i, dummy.matrix);
        mesh.setColorAt(i, new THREE.Color(palette[Math.floor(random() * palette.length)]));
      }
      mesh.instanceMatrix.needsUpdate = true;
      tree.add(mesh);
      return mesh;
    }

    const blossoms = makeCanopy(flowerGeometry, [
      "#fff2ed", "#ffd6df", "#ffb8cc", "#ffe7ec"
    ], true);
    const summerLeaves = makeCanopy(leafGeometry, [
      "#55a457", "#77bb62", "#3f924c", "#9dca70"
    ]);
    const autumnLeaves = makeCanopy(leafGeometry, [
      "#e9aa45", "#cf653e", "#f1ca62", "#b94d3b"
    ]);

    function makeParticles(n, geometry, colors, material) {
      const mesh = new THREE.InstancedMesh(geometry, material, n);
      mesh.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
      for (let i = 0; i < n; i++) {
        mesh.setColorAt(i, new THREE.Color(colors[Math.floor(random() * colors.length)]));
      }
      scene.add(mesh);
      return mesh;
    }

    const fallingCount = 85;
    const falling = makeParticles(
      fallingCount, leafGeometry,
      ["#e7a44b", "#be563b", "#edc45b"],
      new THREE.MeshStandardMaterial({
        color: 0xffffff, side: THREE.DoubleSide,
        transparent: true, opacity: 0, depthWrite: false
      })
    );
    const fallingData = Array.from({ length: fallingCount }, () => ({
      x: range(-2.2, 2.2), z: range(-1.9, 1.9),
      offset: range(0, 5.4), speed: range(0.45, 0.85),
      phase: range(0, Math.PI * 2)
    }));

    const snowCount = 260;
    const snow = makeParticles(
      snowCount, new THREE.SphereGeometry(1, 5, 4), ["#ffffff", "#eaf7ff"],
      new THREE.MeshBasicMaterial({
        color: 0xffffff, transparent: true, opacity: 0, depthWrite: false
      })
    );
    const snowData = Array.from({ length: snowCount }, () => ({
      x: range(-6, 6), z: range(-5, 5),
      offset: range(0, 6), speed: range(0.35, 0.85),
      size: range(0.025, 0.065), phase: range(0, Math.PI * 2)
    }));

    const cloudMaterial = new THREE.MeshBasicMaterial({
      color: 0xffffff, transparent: true, opacity: 0.6, depthWrite: false
    });
    const clouds = [];
    for (const [x, y, z, scale] of [
      [-7, 7.4, -11, 1.1], [5, 8.2, -15, 1.4], [0, 6.9, -18, 0.8]
    ]) {
      const group = new THREE.Group();
      for (let i = 0; i < 4; i++) {
        const puff = new THREE.Mesh(new THREE.SphereGeometry(1, 12, 8), cloudMaterial);
        puff.position.set((i - 1.5) * 0.75, i % 2 ? 0.2 : 0, 0);
        puff.scale.set(0.9, 0.45 + i * 0.045, 0.55);
        group.add(puff);
      }
      group.position.set(x, y, z);
      group.scale.setScalar(scale);
      scene.add(group);
      clouds.push(group);
    }

    const seasons = [
      {
        name: "봄", description: "꽃이 피어납니다",
        top: "#8fc7eb", horizon: "#f9d7d5",
        hill: "#91bd76", plain: "#a6ca88", light: "#fff0d8"
      },
      {
        name: "여름", description: "잎이 무성해집니다",
        top: "#5aafe5", horizon: "#c9edfa",
        hill: "#5caa65", plain: "#73b873", light: "#fff5de"
      },
      {
        name: "가을", description: "낙엽이 흩날립니다",
        top: "#8fb4d1", horizon: "#f4d2ad",
        hill: "#aa875f", plain: "#bc9b70", light: "#ffdcaa"
      },
      {
        name: "겨울", description: "눈이 내립니다",
        top: "#829eb8", horizon: "#e2edf4",
        hill: "#e1eaf0", plain: "#d5e2ec", light: "#e8f3ff"
      }
    ];
    const colors = seasons.map(s => ({
      top: new THREE.Color(s.top),
      horizon: new THREE.Color(s.horizon),
      hill: new THREE.Color(s.hill),
      plain: new THREE.Color(s.plain),
      light: new THREE.Color(s.light)
    }));
    const smooth = t => {
      t = THREE.MathUtils.clamp(t, 0, 1);
      return t * t * (3 - 2 * t);
    };

    const label = document.getElementById("season");
    const description = document.getElementById("description");
    const progress = document.getElementById("progress");
    const dummy = new THREE.Object3D();
    const startTime = performance.now();
    let displayedSeason = -1;

    function animate(now) {
      const elapsed = (now - startTime) / 1000;
      const phase = (elapsed % 40) / 10;
      const current = Math.floor(phase);
      const next = (current + 1) % 4;
      const within = phase - current;
      // 각 계절의 마지막 3초 동안 다음 계절로 부드럽게 전환한다.
      const blend = smooth((within - 0.7) / 0.3);
      const weights = [0, 0, 0, 0];
      weights[current] = 1 - blend;
      weights[next] = blend;

      if (displayedSeason !== current) {
        label.textContent = seasons[current].name;
        description.textContent = seasons[current].description;
        displayedSeason = current;
      }
      progress.style.width = `${within * 100}%`;

      for (const key of ["top", "horizon"]) {
        skyMaterial.uniforms[key === "top" ? "topColor" : "horizonColor"].value
          .copy(colors[current][key]).lerp(colors[next][key], blend);
      }
      hillMaterial.color.copy(colors[current].hill).lerp(colors[next].hill, blend);
      plainMaterial.color.copy(colors[current].plain).lerp(colors[next].plain, blend);
      sunLight.color.copy(colors[current].light).lerp(colors[next].light, blend);
      sun.material.color.copy(sunLight.color);
      scene.fog.color.copy(skyMaterial.uniforms.horizonColor.value);

      blossoms.material.opacity = weights[0];
      summerLeaves.material.opacity = weights[1];
      autumnLeaves.material.opacity = weights[2];
      falling.material.opacity = weights[2];
      snow.material.opacity = weights[3];
      cloudMaterial.opacity = 0.6 - weights[3] * 0.17;

      const t = elapsed;
      for (let i = 0; i < fallingCount; i++) {
        const p = fallingData[i];
        const y = 4.6 - ((t * p.speed + p.offset) % 5.4);
        dummy.position.set(
          p.x + Math.sin(t * 1.3 + p.phase) * 0.48,
          y,
          p.z + Math.cos(t + p.phase) * 0.25
        );
        dummy.rotation.set(t * 1.6 + p.phase, t + p.phase, t * 1.2);
        dummy.scale.set(0.075, 0.17, 0.035);
        dummy.updateMatrix();
        falling.setMatrixAt(i, dummy.matrix);
      }
      falling.instanceMatrix.needsUpdate = true;

      for (let i = 0; i < snowCount; i++) {
        const p = snowData[i];
        dummy.position.set(
          p.x + Math.sin(t * 0.65 + p.phase) * 0.35,
          5.5 - ((t * p.speed + p.offset) % 6),
          p.z
        );
        dummy.rotation.set(0, 0, 0);
        dummy.scale.setScalar(p.size);
        dummy.updateMatrix();
        snow.setMatrixAt(i, dummy.matrix);
      }
      snow.instanceMatrix.needsUpdate = true;

      clouds.forEach((cloud, i) => {
        cloud.position.x += Math.sin(t * 0.15 + i) * 0.0008;
      });
      tree.rotation.y = Math.sin(t * 0.15) * 0.025;

      renderer.render(scene, camera);
    }

    renderer.setAnimationLoop(animate);
    addEventListener("resize", () => {
      camera.aspect = innerWidth / innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(innerWidth, innerHeight);
    });
  </script>
</body>
