<body>
  <div class="season-card" aria-live="polite">
    <small>현재 계절</small>
    <strong id="season-name">봄</strong>
    <div class="track" aria-hidden="true"><span id="progress"></span></div>

  <script type="module">
    import * as THREE from "https://cdn.jsdelivr.net/npm/three@0.160.1/build/three.module.js";

    const scene = new THREE.Scene();
    const camera = new THREE.PerspectiveCamera(43, innerWidth / innerHeight, 0.1, 200);
    camera.position.set(8, 5.7, 13.5);
    camera.lookAt(0, 3.1, 0);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(innerWidth, innerHeight);
    renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
    renderer.outputColorSpace = THREE.SRGBColorSpace;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.45;
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    document.body.appendChild(renderer.domElement);

    // 일정한 난수로 매번 같은 나무 모양을 만듭니다.
    let seed = 271828;
    const random = () => {
      seed = (seed * 1664525 + 1013904223) >>> 0;
      return seed / 4294967296;
    };

    const palettes = [
      { top: "#78bce8", bottom: "#f7d9d8", ground: "#77b96c", hill: "#91c77a", fog: "#e8d9d6" }, // 봄
      { top: "#438fce", bottom: "#c5e8ee", ground: "#4e9952", hill: "#65aa57", fog: "#d4e7dd" }, // 여름
      { top: "#7898bf", bottom: "#f3ca9a", ground: "#aa8b4e", hill: "#bf9d58", fog: "#edd5b8" }, // 가을
      { top: "#718eaf", bottom: "#e2e9f1", ground: "#c6d1d7", hill: "#e5e9e5", fog: "#dce5ec" }  // 겨울
    ].map(p => Object.fromEntries(
      Object.entries(p).map(([key, value]) => [key, new THREE.Color(value)])
    ));

    const skyMaterial = new THREE.ShaderMaterial({
      side: THREE.BackSide,
      depthWrite: false,
      uniforms: {
        topColor: { value: palettes[0].top.clone() },
        bottomColor: { value: palettes[0].bottom.clone() }
      },
      vertexShader: `
        varying vec3 vDirection;
        void main() {
          vDirection = position;
          gl_Position = projectionMatrix * modelViewMatrix * vec4(position, 1.0);
        }
      `,
      fragmentShader: `
        uniform vec3 topColor;
        uniform vec3 bottomColor;
        varying vec3 vDirection;
        void main() {
          float h = smoothstep(-0.2, 0.8, normalize(vDirection).y);
          gl_FragColor = vec4(mix(bottomColor, topColor, h), 1.0);
        }
      `
    });
    const sky = new THREE.Mesh(new THREE.SphereGeometry(90, 32, 16), skyMaterial);
    sky.renderOrder = -1000;
    scene.add(sky);
    scene.fog = new THREE.Fog(palettes[0].fog.clone(), 18, 75);

    scene.add(new THREE.AmbientLight(0xffffff, 1.8));
    const sunlight = new THREE.DirectionalLight(0xfff4dc, 2.5);
    sunlight.position.set(-5, 11, 7);
    sunlight.castShadow = true;
    sunlight.shadow.mapSize.set(1024, 1024);
    sunlight.shadow.camera.left = -13;
    sunlight.shadow.camera.right = 13;
    sunlight.shadow.camera.top = 13;
    sunlight.shadow.camera.bottom = -13;
    sunlight.shadow.normalBias = 0.03;
    scene.add(sunlight);

    const groundMaterial = new THREE.MeshLambertMaterial({ color: palettes[0].ground });
    const ground = new THREE.Mesh(new THREE.PlaneGeometry(200, 200), groundMaterial);
    ground.rotation.x = -Math.PI / 2;
    ground.position.y = -1.55;
    ground.receiveShadow = true;
    scene.add(ground);

    const hillMaterial = new THREE.MeshLambertMaterial({ color: palettes[0].hill });
    const hill = new THREE.Mesh(new THREE.SphereGeometry(1, 64, 32), hillMaterial);
    hill.scale.set(9, 2, 7);
    hill.position.y = -1.55;
    hill.receiveShadow = true;
    scene.add(hill);

    const sunDisc = new THREE.Mesh(
      new THREE.SphereGeometry(1.15, 24, 16),
      new THREE.MeshBasicMaterial({ color: "#fff1cf", transparent: true, opacity: 0.8, fog: false })
    );
    sunDisc.position.set(-15, 16, -32);
    scene.add(sunDisc);

    const bark = new THREE.MeshLambertMaterial({ color: "#635043" });
    const twigTips = [];
    const up = new THREE.Vector3(0, 1, 0);

    function segment(a, b, bottomRadius, topRadius) {
      const direction = new THREE.Vector3().subVectors(b, a);
      const mesh = new THREE.Mesh(
        new THREE.CylinderGeometry(topRadius, bottomRadius, direction.length(), 7),
        bark
      );
      mesh.position.copy(a).add(b).multiplyScalar(0.5);
      mesh.quaternion.setFromUnitVectors(up, direction.normalize());
      mesh.castShadow = true;
      scene.add(mesh);
    }

    const trunk = [
      new THREE.Vector3(0, .37, 0),
      new THREE.Vector3(.08, 1.55, 0),
      new THREE.Vector3(-.02, 2.8, .03),
      new THREE.Vector3(.09, 4.02, 0),
      new THREE.Vector3(-.03, 5.12, -.02),
      new THREE.Vector3(0, 6.05, 0)
    ];
    const radii = [.36, .30, .24, .17, .11, .035];
    for (let i = 0; i < trunk.length - 1; i++) {
      segment(trunk[i], trunk[i + 1], radii[i], radii[i + 1]);
    }

    function branch(start, direction, length, radius, depth) {
      const end = start.clone().addScaledVector(direction, length);
      segment(start, end, radius, Math.max(.012, radius * .68));
      if (depth === 0) {
        twigTips.push(end);
        return;
      }
      for (const side of [-1, 1]) {
        const childDirection = direction.clone()
          .applyAxisAngle(up, side * (.4 + random() * .35))
          .add(new THREE.Vector3((random() - .5) * .15, .27 + random() * .16, (random() - .5) * .15))
          .normalize();
        branch(end, childDirection, length * (.65 + random() * .08), radius * .68, depth - 1);
      }
    }

    for (let level = 1; level <= 4; level++) {
      const count = level === 1 ? 5 : 4;
      for (let j = 0; j < count; j++) {
        const angle = j * Math.PI * 2 / count + level * .73 + random() * .25;
        const direction = new THREE.Vector3(
          Math.cos(angle),
          .43 + random() * .28,
          Math.sin(angle)
        ).normalize();
        branch(
          trunk[level].clone(),
          direction,
          1.45 + random() * .55 + (4 - level) * .12,
          .105 - level * .012,
          2
        );
      }
    }

    const leafGeometry = new THREE.SphereGeometry(1, 6, 5);
    const blossomGeometry = new THREE.IcosahedronGeometry(1, 0);

    function makeFoliage(count, geometry, colors, shape) {
      const material = new THREE.MeshLambertMaterial({
        color: 0xffffff, transparent: true, opacity: 1,
        depthWrite: false, side: THREE.DoubleSide
      });
      const mesh = new THREE.InstancedMesh(geometry, material, count);
      const dummy = new THREE.Object3D();
      const color = new THREE.Color();
      for (let i = 0; i < count; i++) {
        const anchor = twigTips[Math.floor(random() * twigTips.length)];
        const theta = random() * Math.PI * 2;
        const z = random() * 2 - 1;
        const radius = .12 + random() * .72;
        const horizontal = Math.sqrt(1 - z * z);
        dummy.position.set(
          anchor.x + Math.cos(theta) * horizontal * radius,
          anchor.y + z * radius,
          anchor.z + Math.sin(theta) * horizontal * radius
        );
        dummy.rotation.set(random() * 3, random() * 6.28, random() * 3);
        const size = .75 + random() * .65;
        dummy.scale.set(shape[0] * size, shape[1] * size, shape[2] * size);
        dummy.updateMatrix();
        mesh.setMatrixAt(i, dummy.matrix);
        color.set(colors[Math.floor(random() * colors.length)]);
        mesh.setColorAt(i, color);
      }
      mesh.instanceMatrix.needsUpdate = true;
      scene.add(mesh);
      return mesh;
    }

    const blossoms = makeFoliage(
      920, blossomGeometry,
      ["#fff7ec", "#ffe3e4", "#ffcbd8", "#f9aebe"],
      [.105, .105, .105]
    );
    const greenLeaves = makeFoliage(
      1050, leafGeometry,
      ["#4f9b4e", "#69b85c", "#83c665", "#a0cc71"],
      [.11, .22, .07]
    );
    const autumnLeaves = makeFoliage(
      950, leafGeometry,
      ["#d88436", "#e5a642", "#ba5a36", "#e2bd5d", "#b97230"],
      [.115, .22, .07]
    );

    // 언덕의 작은 풀과 봄꽃.
    const grassMaterial = new THREE.MeshLambertMaterial({
      color: "#8bc76a", transparent: true, opacity: 1, depthWrite: false
    });
    const grass = new THREE.InstancedMesh(
      new THREE.ConeGeometry(.035, .25, 3), grassMaterial, 330
    );
    const flowerMaterial = new THREE.MeshLambertMaterial({
      color: 0xffffff, transparent: true, opacity: 1, depthWrite: false
    });
    const meadowFlowers = new THREE.InstancedMesh(
      new THREE.IcosahedronGeometry(.065, 0), flowerMaterial, 130
    );
    const dummy = new THREE.Object3D();
    const flowerColors = ["#fff9e9", "#ffe4d8", "#f7bfd2"];
    for (let i = 0; i < 330; i++) {
      let x, z, q;
      do {
        x = (random() - .5) * 15;
        z = (random() - .5) * 11;
        q = (x / 9) ** 2 + (z / 7) ** 2;
      } while (q >= .82 || (x * x + z * z < 1.2));
      const y = -1.55 + 2 * Math.sqrt(1 - q);
      dummy.position.set(x, y + .1, z);
      dummy.rotation.set((random() - .5) * .35, random() * 6.28, (random() - .5) * .35);
      dummy.scale.setScalar(.65 + random() * .8);
      dummy.updateMatrix();
      grass.setMatrixAt(i, dummy.matrix);
      if (i < 130) {
        dummy.position.y = y + .09;
        dummy.scale.setScalar(.6 + random() * .7);
        dummy.updateMatrix();
        meadowFlowers.setMatrixAt(i, dummy.matrix);
        meadowFlowers.setColorAt(i, new THREE.Color(
          flowerColors[Math.floor(random() * flowerColors.length)]
        ));
      }
    }
    grass.instanceMatrix.needsUpdate = true;
    meadowFlowers.instanceMatrix.needsUpdate = true;
    scene.add(grass, meadowFlowers);

    // 가을 낙엽.
    const fallingMaterial = new THREE.MeshLambertMaterial({
      color: "#e6a244", transparent: true, opacity: 1,
      depthWrite: false, side: THREE.DoubleSide
    });
    const fallingLeaves = new THREE.InstancedMesh(leafGeometry, fallingMaterial, 95);
    const fallingData = Array.from({ length: 95 }, () => ({
      x: (random() - .5) * 6.4,
      z: (random() - .5) * 5,
      offset: random() * 7.5,
      speed: .65 + random() * .65,
      phase: random() * 6.28
    }));
    scene.add(fallingLeaves);

    // 겨울 눈.
    const snowCount = 260;
    const snowPositions = new Float32Array(snowCount * 3);
    const snowData = Array.from({ length: snowCount }, () => ({
      x: (random() - .5) * 18,
      z: (random() - .5) * 15,
      offset: random() * 12,
      speed: .55 + random() * .8,
      phase: random() * 6.28
    }));
    const snowGeometry = new THREE.BufferGeometry();
    snowGeometry.setAttribute("position", new THREE.BufferAttribute(snowPositions, 3));
    const snowMaterial = new THREE.PointsMaterial({
      color: 0xffffff, size: .105, sizeAttenuation: true,
      transparent: true, opacity: 0, depthWrite: false
    });
    const snow = new THREE.Points(snowGeometry, snowMaterial);
    scene.add(snow);

    const seasonNames = ["봄", "여름", "가을", "겨울"];
    const nameElement = document.getElementById("season-name");
    const progressElement = document.getElementById("progress");
    const weights = [0, 0, 0, 0];
    const mixedColor = new THREE.Color();
    let displayedSeason = -1;

    function mixPalette(key, output) {
      output.setRGB(0, 0, 0);
      for (let i = 0; i < 4; i++) {
        output.r += palettes[i][key].r * weights[i];
        output.g += palettes[i][key].g * weights[i];
        output.b += palettes[i][key].b * weights[i];
      }
      return output;
    }

    const startTime = performance.now();
    function animate(now) {
      const elapsed = (now - startTime) / 1000;
      const cycleTime = elapsed % 40;
      const season = Math.floor(cycleTime / 10);
      const withinSeason = cycleTime % 10;

      // 각 계절의 마지막 2.6초 동안 다음 계절과 부드럽게 섞습니다.
      const transition = THREE.MathUtils.smoothstep(withinSeason, 7.4, 10);
      weights.fill(0);
      weights[season] = 1 - transition;
      weights[(season + 1) % 4] += transition;

      if (season !== displayedSeason) {
        displayedSeason = season;
        nameElement.textContent = seasonNames[season];
      }
      progressElement.style.width = `${cycleTime / 40 * 100}%`;

      skyMaterial.uniforms.topColor.value.copy(mixPalette("top", mixedColor));
      skyMaterial.uniforms.bottomColor.value.copy(mixPalette("bottom", mixedColor));
      groundMaterial.color.copy(mixPalette("ground", mixedColor));
      hillMaterial.color.copy(mixPalette("hill", mixedColor));
      scene.fog.color.copy(mixPalette("fog", mixedColor));

      blossoms.material.opacity = weights[0] * .96;
      greenLeaves.material.opacity = Math.min(1, weights[1] + weights[0] * .28 + weights[2] * .06);
      autumnLeaves.material.opacity = weights[2] * .97;
      grassMaterial.opacity = weights[0] + weights[1] + weights[2] * .78;
      meadowFlowers.material.opacity = weights[0];
      fallingMaterial.opacity = weights[2] * .9;
      snowMaterial.opacity = weights[3] * .9;
      blossoms.visible = blossoms.material.opacity > .005;
      greenLeaves.visible = greenLeaves.material.opacity > .005;
      autumnLeaves.visible = autumnLeaves.material.opacity > .005;
      grass.visible = grassMaterial.opacity > .005;
      meadowFlowers.visible = meadowFlowers.material.opacity > .005;
      fallingLeaves.visible = fallingMaterial.opacity > .005;
      snow.visible = snowMaterial.opacity > .005;
      grassMaterial.color.copy(palettes[0].hill).lerp(palettes[2].hill, weights[2] * .75);

      if (fallingLeaves.visible) {
        for (let i = 0; i < fallingData.length; i++) {
          const p = fallingData[i];
          const fall = (elapsed * p.speed + p.offset) % 7.5;
          dummy.position.set(
            p.x + Math.sin(elapsed * 1.1 + p.phase) * .55 + fall * .15,
            7.3 - fall,
            p.z + Math.cos(elapsed * .8 + p.phase) * .3
          );
          dummy.rotation.set(elapsed * 1.6 + p.phase, elapsed * 1.2, elapsed * .9);
          dummy.scale.set(.10, .20, .055);
          dummy.updateMatrix();
          fallingLeaves.setMatrixAt(i, dummy.matrix);
        }
        fallingLeaves.instanceMatrix.needsUpdate = true;
      }

      if (snow.visible) {
        for (let i = 0; i < snowCount; i++) {
          const p = snowData[i];
          snowPositions[i * 3] = p.x + Math.sin(elapsed * .7 + p.phase) * .5;
          snowPositions[i * 3 + 1] = 9 - ((elapsed * p.speed + p.offset) % 12);
          snowPositions[i * 3 + 2] = p.z;
        }
        snowGeometry.attributes.position.needsUpdate = true;
      }

      camera.position.x = 8 + Math.sin(elapsed * .12) * .45;
      camera.lookAt(0, 3.1, 0);
      sky.position.copy(camera.position);
      renderer.render(scene, camera);
      requestAnimationFrame(animate);
    }
    requestAnimationFrame(animate);

    addEventListener("resize", () => {
      camera.aspect = innerWidth / innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(innerWidth, innerHeight);
      renderer.setPixelRatio(Math.min(devicePixelRatio, 2));
    });
  </script>
</body>
