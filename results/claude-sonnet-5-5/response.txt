3D 나무가 사계절을 계속 도는 애니메이션을 하나의 HTML 파일로 만들었습니다. 실행하거나 화면을 직접 확인해 보지는 못했습니다. three.js는 r160 고정 주소로 불러오고, 이미지 파일은 쓰지 않았습니다. 눈송이 질감은 canvas로 그렸습니다.

- **계절 주기**: 계절마다 정확히 10초이고, 한 바퀴는 40초입니다. 화면 구석의 계절 이름은 10초 경계에서 바뀝니다.
- **전환**: 경계 앞뒤 2초씩 총 4초 동안 하늘, 땅, 햇빛, 잎과 꽃의 양, 낙엽, 눈이 함께 서서히 섞입니다. 페이지를 처음 열 때는 전환 없이 봄으로 시작합니다.
- **계절별 모습**:
  - 봄: 분홍 꽃과 연둣빛 새잎, 떨어지는 꽃잎
  - 여름: 무성한 초록 잎
  - 가을: 붉은색, 주황색, 노란색 단풍과 떨어지는 낙엽
  - 겨울: 앙상한 가지와 쌓인 눈, 내리는 눈
- **추가 기능**: 카메라가 나무 주위를 천천히 돌고, 마우스나 터치로 드래그하면 시점을 돌릴 수 있습니다.

function grow(p, dir, len, rad, depth) {

class Faller {
  constructor(geo, mat, n, o) {
    this.n = n; this.o = o;
    this.mesh = new THREE.InstancedMesh(geo, mat, n);
    this.mesh.frustumCulled = false;
    this.st = new Uint8Array(n);
    this.p = new Float32Array(n * 3); this.r = new Float32Array(n * 3); this.rs = new Float32Array(n * 3);
    this.vy = new Float32Array(n); this.ph = new Float32Array(n); this.tm = new Float32Array(n); this.sc = new Float32Array(n);
    for (let i = 0; i < n; i++) { this.mesh.setColorAt(i, WHITE); this.hide(i); }
    this.mesh.instanceMatrix.needsUpdate = true;
    scene.add(this.mesh);
  }
  hide(i) {
    dummy.position.set(0, -50, 0); dummy.rotation.set(0, 0, 0); dummy.scale.setScalar(0.0001);
    dummy.updateMatrix(); this.mesh.setMatrixAt(i, dummy.matrix);
  }
  spawn(i) {
    if (!this.o.spawn(this, i)) return false;
    const o = this.o, k = i * 3;
    this.vy[i] = o.vmin + Math.random() * o.vrange;
    this.ph[i] = Math.random() * 6.28;
    for (let j = 0; j < 3; j++) { this.r[k + j] = Math.random() * 6.28; this.rs[k + j] = (Math.random() - 0.5) * 2 * o.spin; }
    this.st[i] = 1;
    return true;
  }
  update(dt, t, inten, wx, wz) {
    const o = this.o, n = this.n, target = inten * n;
    let cd = false;
    for (let i = 0; i < n; i++) {
      let s = this.st[i];
      const k = i * 3;
      if (s === 0) {
        if (i < target && Math.random() < o.rate * dt && this.spawn(i)) { s = 1; cd = true; } else continue;
      }
      let sc = this.sc[i];
      if (s === 1) {
        this.p[k + 1] -= this.vy[i] * dt;
        this.p[k] += (wx * o.drift + Math.sin(t * o.fx + this.ph[i]) * o.sway) * dt;
        this.p[k + 2] += (wz * o.drift + Math.cos(t * o.fx * 0.8 + this.ph[i]) * o.sway) * dt;
        this.r[k] += this.rs[k] * dt; this.r[k + 1] += this.rs[k + 1] * dt; this.r[k + 2] += this.rs[k + 2] * dt;
        const gy = hillH(this.p[k], this.p[k + 2]) + 0.04;
        if (this.p[k + 1] <= gy) {
          this.p[k + 1] = gy; this.st[i] = 2;
          this.tm[i] = o.rest * (0.6 + Math.random() * 0.8);
          this.r[k] = -Math.PI / 2; this.r[k + 1] = (Math.random() - 0.5) * 0.3; this.r[k + 2] = Math.random() * 6.28;
        }
      } else {
        this.tm[i] -= dt;
        if (this.tm[i] <= 0) { this.st[i] = 0; this.hide(i); continue; }
        sc *= Math.min(1, this.tm[i] / 1.2);
      }
      dummy.position.set(this.p[k], this.p[k + 1], this.p[k + 2]);
      dummy.rotation.set(this.r[k], this.r[k + 1], this.r[k + 2]);
      dummy.scale.setScalar(Math.max(sc, 0.0001));
      dummy.updateMatrix();
      this.mesh.setMatrixAt(i, dummy.matrix);
    }
    this.mesh.instanceMatrix.needsUpdate = true;
    if (cd) this.mesh.instanceColor.needsUpdate = true;
  }
}
