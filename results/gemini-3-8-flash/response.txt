            const canvas = document.createElement('canvas');
            canvas.width = 64;
            canvas.height = 64;
            const ctx = canvas.getContext('2d');
            ctx.clearRect(0, 0, 64, 64);
            ctx.fillStyle = '#ffffff';
            ctx.beginPath();
            ctx.moveTo(32, 4);
            ctx.bezierCurveTo(56, 16, 56, 48, 32, 60);
            ctx.bezierCurveTo(8, 48, 8, 16, 32, 4);
            ctx.closePath();
            ctx.fill();
            return new THREE.CanvasTexture(canvas);

            grad.addColorStop(0.4, 'rgba(255, 255, 255, 0.85)');
            grad.addColorStop(1, 'rgba(255, 255, 255, 0)');
            ctx.fillStyle = grad;
            ctx.beginPath();
            ctx.arc(32, 32, 30, 0, Math.PI * 2);
            ctx.fill();
            return new THREE.CanvasTexture(canvas);
        }

        function createBranch(start, dir, length, radius, depth) {
            const end = start.clone().add(dir.clone().multiplyScalar(length));
            const mid = start.clone().add(end).multiplyScalar(0.5);

            // 가지 실린더
            const branchGeom = new THREE.CylinderGeometry(radius * 0.68, radius, length, 8);

            const quat = new THREE.Quaternion().setFromUnitVectors(up, dir.clone().normalize());
            branchMesh.setRotationFromQuaternion(quat);
            branchMesh.position.copy(mid);
            branchMesh.castShadow = true;
            branchMesh.receiveShadow = true;

                    dir.z * 0.6 + Math.sin(angle) * spread

                createBranch(
                    end,
                    newDir,
                    length * (0.68 + Math.random() * 0.14),
                    radius * 0.65,
                    depth - 1

            }

        const curParticleColor = new THREE.Color();
        const tempLeafColor = new THREE.Color();

        function animate() {
            requestAnimationFrame(animate);

            const elapsedTime = clock.getElapsedTime();
            const cycleTime = elapsedTime % TOTAL_CYCLE;

            // 계절 인덱스 계산 (0: 봄, 1: 여름, 2: 가을, 3: 겨울)
            const seasonIndex = Math.floor(cycleTime / SEASON_DURATION);
            const nextSeasonIndex = (seasonIndex + 1) % 4;

            // 계절 내 진행률 (0.0 ~ 1.0)
            const seasonProgress = (cycleTime % SEASON_DURATION) / SEASON_DURATION;

            // 계절간 부드러운 전환 블렌드 가중치 (끝부분 3초 동안 서서히 다음 계절로 전환)
            const transitionStart = 0.7; // 7초부터 전환 시작
            let blend = 0;
            if (seasonProgress > transitionStart) {
                const rawBlend = (seasonProgress - transitionStart) / (1.0 - transitionStart);
                blend = smoothstep(rawBlend);
            }

            const cCur = SEASON_CONFIGS[seasonIndex];
            const cNext = SEASON_CONFIGS[nextSeasonIndex];

            // 1) 색상 및 속성 보간
            curSky.copy(cCur.skyColor).lerp(cNext.skyColor, blend);
            curFog.copy(cCur.fogColor).lerp(cNext.fogColor, blend);
            curGround.copy(cCur.groundColor).lerp(cNext.groundColor, blend);
            curSun.copy(cCur.sunColor).lerp(cNext.sunColor, blend);
            curHemiSky.copy(cCur.hemiSky).lerp(cNext.hemiSky, blend);
            curHemiGround.copy(cCur.hemiGround).lerp(cNext.hemiGround, blend);
            curLeafColor.copy(cCur.leafBaseColor).lerp(cNext.leafBaseColor, blend);
            curParticleColor.copy(cCur.particleColor).lerp(cNext.particleColor, blend);

            const curSunIntensity = THREE.MathUtils.lerp(cCur.sunIntensity, cNext.sunIntensity, blend);
            const curSnowCover = THREE.MathUtils.lerp(cCur.snowCover, cNext.snowCover, blend);
            const curLeafScale = THREE.MathUtils.lerp(cCur.leafScale, cNext.leafScale, blend);
            const curParticleOpacity = THREE.MathUtils.lerp(cCur.particleOpacity, cNext.particleOpacity, blend);
            const curParticleSpeed = THREE.MathUtils.lerp(cCur.particleSpeed, cNext.particleSpeed, blend);

            // 2) 씬 환경 업데이트
            scene.background = curSky;
            scene.fog.color = curFog;
            renderer.setClearColor(curFog);

            hillMaterial.color.copy(curGround);
            dirLight.color.copy(curSun);
            dirLight.intensity = curSunIntensity;
            hemiLight.color.copy(curHemiSky);
            hemiLight.groundColor.copy(curHemiGround);

            // 겨울 가지 눈 쌓임 효과
            snowCapMaterial.opacity = curSnowCover;

            // 3) UI 업데이트
            uiTitle.textContent = cCur.name;
            uiBadge.textContent = cCur.badge;
            uiDesc.textContent = cCur.desc;
            uiProgress.style.width = `${(seasonProgress * 100).toFixed(1)}%`;

            // 4) 나뭇잎 크기 및 바람에 흔들리는 애니메이션
            for (let i = 0; i < totalLeaves; i++) {
                const data = leafData[i];
                if (!data) continue;

                // 잎 크기: 겨울에는 0으로 축소되어 앙상한 가지만 남음
                let scale = data.baseScale * curLeafScale;

                // 봄에는 작게 싹터서 만개하는 효과
                if (seasonIndex === 0 && seasonProgress < 0.3) {
                    scale *= (seasonProgress / 0.3) * 0.7 + 0.3;
                }

                if (scale < 0.001) {
                    scale = 0.0001;
                }

                // 부드러운 바람 흔들림
                const sway = Math.sin(elapsedTime * data.swaySpeed + data.swayOffset) * 0.12;

                dummy.position.copy(data.pos);
                dummy.rotation.set(
                    data.baseRot.x + sway,
                    data.baseRot.y + sway * 0.8,
                    data.baseRot.z + sway * 0.5
                );
                dummy.scale.setScalar(scale);
                dummy.updateMatrix();
                leafMesh.setMatrixAt(i, dummy.matrix);

                // 개별 잎 색상 지터
                tempLeafColor.copy(curLeafColor);
                tempLeafColor.offsetHSL(data.hueVariation, 0, data.lightVariation);
                leafMesh.setColorAt(i, tempLeafColor);
            }
            leafMesh.instanceMatrix.needsUpdate = true;
            if (leafMesh.instanceColor) leafMesh.instanceColor.needsUpdate = true;

            // 5) 대기 파티클 (꽃잎/낙엽/눈송이) 낙하 및 흩날림 시뮬레이션
            const posAttr = particleGeom.attributes.position;
            const colAttr = particleGeom.attributes.color;
            const posArr = posAttr.array;
            const colArr = colAttr.array;

            particleMaterial.opacity = curParticleOpacity;

            for (let i = 0; i < particleCount; i++) {
                const vel = particleVelocities[i];
                const i3 = i * 3;

                // 낙하 속도
                posArr[i3 + 1] -= vel.vy * curParticleSpeed * 0.08;

                // 바람에 의한 회전 및 좌우 흔들림
                posArr[i3 + 0] += Math.sin(elapsedTime * vel.vxOsc + vel.phase) * 0.035;
                posArr[i3 + 2] += Math.cos(elapsedTime * vel.vzOsc + vel.phase) * 0.025;

                // 바닥이나 범위 아래로 떨어지면 위에서 재생성
                if (posArr[i3 + 1] < 0.2) {
                    posArr[i3 + 1] = 13 + Math.random() * 4;
                    posArr[i3 + 0] = (Math.random() - 0.5) * 22;
                    posArr[i3 + 2] = (Math.random() - 0.5) * 22;
                }

                // 파티클 색상 보간
                colArr[i3 + 0] = curParticleColor.r;
                colArr[i3 + 1] = curParticleColor.g;
                colArr[i3 + 2] = curParticleColor.b;
            }
            posAttr.needsUpdate = true;
            colAttr.needsUpdate = true;

            // 6) 카메라 자연스러운 자동 회전 (드래그하지 않을 때 은은하게 회전)
            if (!isDragging) {
                cameraAngleX += 0.0018;
            }

            const targetY = 4.2;
            camera.position.x = Math.sin(cameraAngleX) * Math.cos(cameraAngleY) * cameraDistance;
            camera.position.y = targetY + Math.sin(cameraAngleY) * cameraDistance * 0.7;
            camera.position.z = Math.cos(cameraAngleX) * Math.cos(cameraAngleY) * cameraDistance;
            camera.lookAt(0, 3.8, 0);

            // 렌더링
            renderer.render(scene, camera);
        }

        // 실행 시작
        animate();
    </script>
</body>
</html>
