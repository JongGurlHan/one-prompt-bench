아래 코드를 `index.html`로 저장한 뒤 브라우저에서 열면 됩니다. three.js를 고정 버전의 CDN 주소에서 불러오므로 처음 실행할 때 인터넷 연결이 필요합니다.

```html
<!DOCTYPE html>
<html lang="ko">
<head>

  <title>사계절 나무</title>
  <style>
    * { box-sizing: border-box; }

    html, body {

<body>
  <div class="label">
    <small>FOUR SEASONS</small>
    <strong id="season">봄</strong>
  </div>
  <div class="note">40초에 한 번, 계절이 돌아옵니다</div>

  <!-- 고정 버전의 three.js만 사용합니다. -->
  <script src="https://cdn.jsdelivr.net/npm/three@0.160.0/build/three.min.js"></script>
  <script>
    (() => {
      const THREE = window.THREE;
      const label = document.getElementById("season");

      if (!THREE) {
        label.textContent = "화면을 불러오지 못했습니다";
        return;
      }

      // 재현 가능한 난수: 새로고침해도 같은 나무를 만듭니다.
      let seed = 82473;
      function random() {
        seed = (seed * 1664525 + 1013904223) >>> 0;
        return seed / 4294967296;
      }
      function range(min, max) {
        return min + (max - min) * random();
      }

      const scene = new THREE.Scene();
      scene.background = new THREE.Color();
      scene.fog = new THREE.Fog(0xffffff, 16, 38);

      const camera = new THREE.PerspectiveCamera(
        38, window.innerWidth / window.innerHeight, 0.1, 100
      );
      camera.position.set(9, 6.2, 13);
      camera.lookAt(0, 2.5, 0);

      const renderer = new THREE.WebGLRenderer({ antialias: true });
      renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      renderer.setSize(window.innerWidth, window.innerHeight);
      renderer.outputColorSpace = THREE.SRGBColorSpace;
      renderer.shadowMap.enabled = true;
      renderer.shadowMap.type = THREE.PCFSoftShadowMap;
      document.body.appendChild(renderer.domElement);

      const hemisphere = new THREE.HemisphereLight(0xffffff, 0x6c7757, 2.0);
      scene.add(hemisphere);

      const sun = new THREE.DirectionalLight(0xffffff, 2.2);
      sun.position.set(-5, 10, 7);
      sun.castShadow = true;
      sun.shadow.mapSize.set(1024, 1024);
      sun.shadow.camera.left = -10;
      sun.shadow.camera.right = 10;
      sun.shadow.camera.top = 10;
      sun.shadow.camera.bottom = -10;
      sun.shadow.camera.near = 0.5;
      sun.shadow.camera.far = 30;
      sun.shadow.bias = -0.0003;
      scene.add(sun);

      // 완만한 언덕: 구의 윗부분을 지면 위에 드러냅니다.
      const groundMaterial = new THREE.MeshStandardMaterial({
        color: 0x83b969,
        roughness: 1
      });
      const hill = new THREE.Mesh(
        new THREE.SphereGeometry(1, 64, 32),
        groundMaterial
      );
      hill.scale.set(11, 2.2, 11);
      hill.position.y = -2.15;
      hill.receiveShadow = true;
      scene.add(hill);

      // 먼 지면이 화면 가장자리까지 이어지도록 합니다.
      const farGround = new THREE.Mesh(
        new THREE.PlaneGeometry(200, 200),
        groundMaterial
      );
      farGround.rotation.x = -Math.PI / 2;
      farGround.position.y = -1.65;
      farGround.receiveShadow = true;
      scene.add(farGround);

      const tree = new THREE.Group();
      scene.add(tree);

      const bark = new THREE.MeshStandardMaterial({
        color: 0x69503b,
        roughness: 1
      });
      const branchGeometry = new THREE.CylinderGeometry(1, 1, 1, 7);
      const endpoints = [];
      const up = new THREE.Vector3(0, 1, 0);

      function addBranch(start, direction, length, radius, depth) {
        const end = start.clone().addScaledVector(direction, length);
        const midpoint = start.clone().add(end).multiplyScalar(0.5);
        const branch = new THREE.Mesh(branchGeometry, bark);
        branch.position.copy(midpoint);
        branch.quaternion.setFromUnitVectors(up, direction);
        branch.scale.set(radius * 0.72, length, radius);
        branch.castShadow = true;
        branch.receiveShadow = true;
        tree.add(branch);

        if (depth === 0) {
          endpoints.push(end);
          return;
        }

        const count = depth === 3 ? 4 : 3;
        const baseAngle = range(0, Math.PI * 2);

        for (let i = 0; i < count; i++) {
          const angle = baseAngle + i * Math.PI * 2 / count + range(-0.35, 0.35);
          const outward = new THREE.Vector3(
            Math.cos(angle),
            range(0.35, 0.8),
            Math.sin(angle)
          ).normalize();

          const nextDirection = direction.clone()
            .multiplyScalar(0.42)
            .addScaledVector(outward, 0.85)
            .normalize();

          const branchStart = start.clone().lerp(end, range(0.68, 0.98));
          addBranch(
            branchStart,
            nextDirection,
            length * range(0.62, 0.76),
            radius * range(0.51, 0.62),
            depth - 1
          );
        }
      }

      addBranch(
        new THREE.Vector3(0, -0.02, 0),
        new THREE.Vector3(0, 1, 0),
        2.3,
        0.24,
        3
      );

      // 잎과 꽃은 가지 끝 주변에 배치합니다.
      const leafPositions = [];
      const blossomPositions = [];

      for (const end of endpoints) {
        for (let i = 0; i < 11; i++) {
          leafPositions.push({
            position: end.clone().add(new THREE.Vector3(
              range(-0.53, 0.53),
              range(-0.25, 0.55),
              range(-0.53, 0.53)
            )),
            size: range(0.17, 0.29),
            rotation: range(0, Math.PI * 2),
            phase: range(0, Math.PI * 2),
            colorMix: random()
          });
        }

        for (let i = 0; i < 4; i++) {
          blossomPositions.push({
            position: end.clone().add(new THREE.Vector3(
              range(-0.47, 0.47),
              range(-0.1, 0.6),
              range(-0.47, 0.47)
            )),
            size: range(0.1, 0.16),
            phase: range(0, Math.PI * 2)
          });
        }
      }

      const leafMaterial = new THREE.MeshStandardMaterial({
        color: 0xffffff,
        roughness: 0.9,
        side: THREE.DoubleSide
      });
      const leaves = new THREE.InstancedMesh(
        new THREE.SphereGeometry(1, 6, 5),
        leafMaterial,
        leafPositions.length
      );
      leaves.castShadow = true;
      leaves.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
      leaves.frustumCulled = false;
      scene.add(leaves);

      const blossoms = new THREE.InstancedMesh(
        new THREE.IcosahedronGeometry(1, 0),
        new THREE.MeshStandardMaterial({
          color: 0xffc8dc,
          roughness: 0.9
        }),
        blossomPositions.length
      );
      blossoms.castShadow = true;
      blossoms.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
      blossoms.frustumCulled = false;
      scene.add(blossoms);

      // 낙엽과 눈은 각각 별도의 입자로 표현합니다.
      const fallingLeafCount = 80;
      const snowCount = 180;
      const fallingLeafData = [];
      const snowData = [];

      for (let i = 0; i < fallingLeafCount; i++) {
        fallingLeafData.push({
          x: range(-4, 4),
          z: range(-3.5, 3.5),
          offset: range(0, 8),
          speed: range(0.45, 0.9),
          phase: range(0, Math.PI * 2),
          size: range(0.06, 0.12)
        });
      }

      for (let i = 0; i < snowCount; i++) {
        snowData.push({
          x: range(-7, 7),
          z: range(-6, 6),
          offset: range(0, 10),
          speed: range(0.35, 0.7),
          phase: range(0, Math.PI * 2),
          size: range(0.025, 0.065)
        });
      }

      const fallingLeaves = new THREE.InstancedMesh(
        new THREE.SphereGeometry(1, 5, 4),
        new THREE.MeshStandardMaterial({ color: 0xd77b32, roughness: 1 }),
        fallingLeafCount
      );
      fallingLeaves.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
      fallingLeaves.frustumCulled = false;
      scene.add(fallingLeaves);

      const snow = new THREE.InstancedMesh(
        new THREE.SphereGeometry(1, 5, 4),
        new THREE.MeshBasicMaterial({ color: 0xffffff }),
        snowCount
      );
      snow.instanceMatrix.setUsage(THREE.DynamicDrawUsage);
      snow.frustumCulled = false;
      scene.add(snow);

      const seasons = [
        {
          name: "봄",
          sky: 0xb9dff1,
          ground: 0x8fc878,
          leafA: 0x8fc96b,
          leafB: 0xb6dc87,
          leafScale: 0.78,
          flowers: 1,
          falling: 0,
          snow: 0,
          light: 2.2
        },
        {
          name: "여름",
          sky: 0x8bcaf0,
          ground: 0x65a55a,
          leafA: 0x398b46,
          leafB: 0x69b858,
          leafScale: 1.25,
          flowers: 0,
          falling: 0,
          snow: 0,
          light: 2.5
        },
        {
          name: "가을",
          sky: 0xeac9a4,
          ground: 0xb59a5c,
          leafA: 0xc94f2c,
          leafB: 0xeab34b,
          leafScale: 0.95,
          flowers: 0,
          falling: 1,
          snow: 0,
          light: 2.0
        },
        {
          name: "겨울",
          sky: 0xc6d9e4,
          ground: 0xe6eeee,
          leafA: 0x8b765e,
          leafB: 0xb0a58d,
          leafScale: 0,
          flowers: 0,
          falling: 0,
          snow: 1,
          light: 1.6
        }
      ];

      const tempObject = new THREE.Object3D();
      const colorA = new THREE.Color();
      const colorB = new THREE.Color();
      const leafColor = new THREE.Color();
      const skyColor = new THREE.Color();
      const groundColor = new THREE.Color();

      function mix(a, b, t) {
        return a + (b - a) * t;
      }

      function smoothstep(t) {
        t = THREE.MathUtils.clamp(t, 0, 1);
        return t * t * (3 - 2 * t);
      }

      function mixColor(out, a, b, t) {
        return out.setHex(a).lerp(new THREE.Color(b), t);
      }

      function setInstance(mesh, index, position, scale, rotation) {
        tempObject.position.copy(position);
        tempObject.rotation.set(
          rotation ? rotation.x : 0,
          rotation ? rotation.y : 0,
          rotation ? rotation.z : 0
        );
        tempObject.scale.set(scale.x, scale.y, scale.z);
        tempObject.updateMatrix();
        mesh.setMatrixAt(index, tempObject.matrix);
      }

      const clock = new THREE.Clock();

      function animate() {
        requestAnimationFrame(animate);

        const time = clock.getElapsedTime();
        const cycleTime = time % 40;
        const index = Math.floor(cycleTime / 10);
        const localTime = cycleTime % 10;
        const current = seasons[index];
        const next = seasons[(index + 1) % seasons.length];

        // 각 계절의 마지막 3초 동안 다음 계절로 부드럽게 섞입니다.
        const transition = smoothstep((localTime - 7) / 3);
        label.textContent = current.name;

        mixColor(skyColor, current.sky, next.sky, transition);
        mixColor(groundColor, current.ground, next.ground, transition);
        scene.background.copy(skyColor);
        scene.fog.color.copy(skyColor);
        groundMaterial.color.copy(groundColor);
        sun.intensity = mix(current.light, next.light, transition);

        const leafScale = mix(current.leafScale, next.leafScale, transition);
        const flowerScale = mix(current.flowers, next.flowers, transition);
        const fallingAmount = mix(current.falling, next.falling, transition);
        const snowAmount = mix(current.snow, next.snow, transition);

        mixColor(colorA, current.leafA, next.leafA, transition);
        mixColor(colorB, current.leafB, next.leafB, transition);

        leafPositions.forEach((leaf, i) => {
          const sway = Math.sin(time * 1.3 + leaf.phase) * 0.035;
          const size = leaf.size * leafScale;
          tempObject.position.copy(leaf.position);
          tempObject.position.x += sway;
          tempObject.rotation.set(0.3, leaf.rotation, sway * 2);
          tempObject.scale.set(size, size * 0.65, size * 0.75);
          tempObject.updateMatrix();
          leaves.setMatrixAt(i, tempObject.matrix);

          leafColor.copy(colorA).lerp(colorB, leaf.colorMix);
          leaves.setColorAt(i, leafColor);
        });
        leaves.instanceMatrix.needsUpdate = true;
        if (leaves.instanceColor) leaves.instanceColor.needsUpdate = true;

        blossomPositions.forEach((flower, i) => {
          const size = flower.size * flowerScale;
          tempObject.position.copy(flower.position);
          tempObject.position.x += Math.sin(time * 1.3 + flower.phase) * 0.035;
          tempObject.rotation.set(0, 0, 0);
          tempObject.scale.setScalar(size);
          tempObject.updateMatrix();
          blossoms.setMatrixAt(i, tempObject.matrix);
        });
        blossoms.instanceMatrix.needsUpdate = true;

        fallingLeafData.forEach((particle, i) => {
          const y = 5.7 - ((time * particle.speed + particle.offset) % 6.2);
          const x = particle.x + Math.sin(time * 1.2 + particle.phase) * 0.45;
          const size = particle.size * fallingAmount;
          setInstance(
            fallingLeaves,
            i,
            new THREE.Vector3(x, y, particle.z),
            new THREE.Vector3(size, size * 0.3, size * 0.65),
            { x: time + particle.phase, y: particle.phase, z: time * 0.7 }
          );
        });
        fallingLeaves.instanceMatrix.needsUpdate = true;

        snowData.forEach((particle, i) => {
          const y = 7 - ((time * particle.speed + particle.offset) % 8);
          const x = particle.x + Math.sin(time * 0.7 + particle.phase) * 0.3;
          const size = particle.size * snowAmount;
          setInstance(
            snow,
            i,
            new THREE.Vector3(x, y, particle.z),
            new THREE.Vector3(size, size, size)
          );
        });
        snow.instanceMatrix.needsUpdate = true;

        renderer.render(scene, camera);
      }

      window.addEventListener("resize", () => {
        camera.aspect = window.innerWidth / window.innerHeight;
        camera.updateProjectionMatrix();
        renderer.setSize(window.innerWidth, window.innerHeight);
        renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
      });

      animate();
    })();
  </script>
</body>
</html>
```
