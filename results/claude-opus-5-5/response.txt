function mulberry32(a) {
  return function () {
    a |= 0; a = (a + 0x6D2B79F5) | 0;
    let t = Math.imul(a ^ (a >>> 15), 1 | a);
    t = (t + Math.imul(t ^ (t >>> 7), 61 | t)) ^ t;
    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
  };
}

function groundH(x, z) {
  const r2 = x * x + z * z;
  const r = Math.sqrt(r2);
  let h = 2.4 * Math.exp(-r2 / 90);
  h += 0.35 * Math.sin(x * 0.21 + 1.3) * Math.cos(z * 0.17 - 0.4) * Math.min(1, r / 12);
  const a = Math.atan2(z, x);
  h += Math.exp(-((r - 85) * (r - 85)) / 500) * (6 + 3 * Math.sin(a * 5 + 1) + 2 * Math.sin(a * 11 + 2));
  return h;
}

const autumnB = { ...autumnA, skyTop: 0x6d8bb0, skyBottom: 0xeccba0, ground: 0x8f7038, cloud: 0xf2e4d4, sunI: 2.5, leaf: 0.3, fall: 1 };
const winterA = {
  skyTop: 0x8d9db3, skyBottom: 0xdfe5ec, ground: 0xe6ebf0, sun: 0xe2ebff, cloud: 0xd8dee6,
  sunI: 2.0, hemiI: 2.0, leaf: 0, blossom: 0, green: 1, autumn: 1, fall: 0, petal: 0,
  snow: 1, snowCover: 1, flowers: 0
};
const winterB = { ...winterA, autumn: 0, green: 0 };

const SA = mk(springA), SBk = mk(springB), SU = mk(summer), AA = mk(autumnA), AB = mk(autumnB), WA = mk(winterA), WB = mk(winterB);
// 시간(초) : 상태. 각 경계 근처에서 부드럽게 넘어가도록 배치
const KF = [
  [1, SA], [7.5, SBk], [11, SU], [17.5, SU], [21, AA], [27.5, AB],
  [31, WA], [35, WB], [37.5, WB], [41, SA]
];
const P = mk(springA);

let firstLoop = true;
function sample(t) {
  const tt = t < 1 ? (firstLoop ? 1 : t + 40) : t;
  let i = 0;
  while (i < KF.length - 2 && tt >= KF[i + 1][0]) i++;
  const [t0, a] = KF[i];
  const [t1, b] = KF[i + 1];
  const f = smooth((tt - t0) / (t1 - t0));
  for (const k of COLOR_KEYS) P[k].copy(a[k]).lerp(b[k], f);
  for (const k of NUM_KEYS) P[k] = a[k] + (b[k] - a[k]) * f;
}

/* ---------------- 환경 적용 ---------------- */
const hemiSkyTmp = new THREE.Color();
function applyEnvironment() {
  skyUniforms.topColor.value.copy(P.skyTop);
  skyUniforms.bottomColor.value.copy(P.skyBottom);
  scene.fog.color.copy(P.skyBottom);
  groundMat.color.copy(P.ground);
  sun.color.copy(P.sun);
  sun.intensity = P.sunI;
  hemiSkyTmp.copy(P.skyBottom).lerp(P.skyTop, 0.4);
  hemi.color.copy(hemiSkyTmp);
  hemi.groundColor.copy(P.ground).multiplyScalar(0.55);
  hemi.intensity = P.hemiI;
  branchMat.color.copy(barkColor).lerp(frostBark, P.snowCover * 0.45);
  cloudMat.color.copy(P.cloud);
  cloudMat.emissive.copy(P.cloud).multiplyScalar(0.35);
  snowMat.opacity = P.snow;
  snow.visible = P.snow > 0.01;
}

/* ---------------- 업데이트 함수들 ---------------- */
const springGreen = new THREE.Color(0xa8d65e);
const summerGreen = new THREE.Color(0x3d8a2d);
const greenTmp = new THREE.Color();

function updateLeaves(T) {
  greenTmp.copy(springGreen).lerp(summerGreen, P.green);
  const grow = 0.55 + 0.45 * clamp01(P.leaf / 0.8);
  for (let i = 0; i < NL; i++) {
    const d = leafData[i];
    const g = clamp01((P.leaf - d.thr) / 0.12);
    leafG[i] = g;
    if (g <= 0) {
      leaves.setMatrixAt(i, ZERO);
    } else {
      eul.set(d.rx + Math.sin(T * 2.6 + d.ph) * 0.15, d.ry, d.rz + Math.cos(T * 2.1 + d.ph) * 0.1);
      q.setFromEuler(eul);
      sc.setScalar(d.s * g * grow);
      m4.compose(d.p, q, sc);
      leaves.setMatrixAt(i, m4);
    }
    const a = smooth((P.autumn - d.ath) / 0.3);
    cTmp.copy(greenTmp).multiplyScalar(d.v).lerp(d.aut, a);
    leaves.setColorAt(i, cTmp);
  }
  leaves.instanceMatrix.needsUpdate = true;
  leaves.instanceColor.needsUpdate = true;
}

function updateBlossoms(T) {
  for (let i = 0; i < NB; i++) {
    const d = blossomData[i];
    const g = clamp01((P.blossom - d.thr) / 0.2);
    blossomG[i] = g;
    if (g <= 0) {
      blossoms.setMatrixAt(i, ZERO);
    } else {
      eul.set(d.rx + Math.sin(T * 2.2 + d.ph) * 0.12, d.ry, d.rz);
      q.setFromEuler(eul);
      sc.setScalar(d.s * g);
      m4.compose(d.p, q, sc);
      blossoms.setMatrixAt(i, m4);
    }
  }
  blossoms.instanceMatrix.needsUpdate = true;
}

function updateFieldFlowers(T) {
  for (let i = 0; i < fieldData.length; i++) {
    const d = fieldData[i];
    const g = clamp01((P.flowers - d.thr) / 0.15);
    if (g <= 0) {
      fieldFlowers.setMatrixAt(i, ZERO);
    } else {
      eul.set(d.rx + Math.sin(T * 1.5 + d.rz) * 0.08, 0, d.rz);
      q.setFromEuler(eul);
      sc.setScalar(d.s * g);
      m4.compose(d.p, q, sc);
      fieldFlowers.setMatrixAt(i, m4);
    }
  }
  fieldFlowers.instanceMatrix.needsUpdate = true;
}

let leafAcc = 0, petalAcc = 0;
function updateFalling(T, dt, wind) {
  leafAcc += P.fall * 32 * dt;
  while (leafAcc >= 1) { leafAcc -= 1; trySpawnLeaf(); }
  petalAcc += P.petal * 16 * dt;
  while (petalAcc >= 1) { petalAcc -= 1; trySpawnPetal(); }

  const rest = P.snowCover > 0.4 ? 1.2 : 5;
  for (let i = 0; i < FN; i++) {
    const st = fState[i];
    if (st === 0) continue;
    const o = i * 3;
    let scale;
    if (st === 1) {
      fp[o + 1] -= fSpeed[i] * dt;
      fp[o] += (wind * 0.9 + Math.sin(T * 1.6 + fPh[i]) * 0.7) * dt;
      fp[o + 2] += (Math.cos(T * 1.2 + fPh[i]) * 0.6 + wind * 0.3) * dt;
      fr[o] += frv[o] * dt;
      fr[o + 1] += frv[o + 1] * dt;
      fr[o + 2] += frv[o + 2] * dt;
      const gh = groundH(fp[o], fp[o + 2]) + 0.04;
      if (fp[o + 1] <= gh) {
        fp[o + 1] = gh;
        fState[i] = 2;
        fTimer[i] = 0;
        fr[o] = -Math.PI / 2;
        fr[o + 1] = 0;
      }
      scale = fSize[i];
    } else {
      fTimer[i] += dt;
      let k = 1;
      if (fTimer[i] > rest) k = 1 - (fTimer[i] - rest) / 1.5;
      if (k <= 0) {
        fState[i] = 0;
        falling.setMatrixAt(i, ZERO);
        continue;
      }
      scale = fSize[i] * k;
    }
    eul.set(fr[o], fr[o + 1], fr[o + 2]);
    q.setFromEuler(eul);
    v3.set(fp[o], fp[o + 1], fp[o + 2]);
    sc.setScalar(scale);
    m4.compose(v3, q, sc);
    falling.setMatrixAt(i, m4);
  }
  falling.instanceMatrix.needsUpdate = true;
}

function updateSnow(T, dt, wind) {
  if (!snow.visible) return;
  for (let i = 0; i < SN; i++) {
    const o = i * 3;
    snowPos[o + 1] -= snowSpd[i] * dt;
    snowPos[o] += (Math.sin(T * 0.6 + snowPh[i]) * 0.35 + wind * 0.4) * dt;
    snowPos[o + 2] += Math.cos(T * 0.5 + snowPh[i]) * 0.3 * dt;
    if (snowPos[o + 1] < -1) {
      snowPos[o + 1] = STOP + Math.random() * 3;
      snowPos[o] = (Math.random() * 2 - 1) * SB;
      snowPos[o + 2] = (Math.random() * 2 - 1) * SB;
    }
    if (snowPos[o] > SB) snowPos[o] -= SB * 2;
    else if (snowPos[o] < -SB) snowPos[o] += SB * 2;
  }
  snowGeo.attributes.position.needsUpdate = true;
}

**구성 설명**

- **계절 흐름**: 40초를 한 바퀴로 하고 봄·여름·가을·겨울에 10초씩 씁니다. 하늘·땅·조명·잎·꽃·낙엽·눈 같은 값을 시간 키프레임으로 정해 두고, 그 사이를 부드러운 곡선(smoothstep)으로 보간합니다. 그래서 계절이 끊기지 않고 서서히 넘어갑니다. 처음 열면 눈이 남지 않은 봄에서 시작합니다.
- **나무**: 정해진 시드의 난수로 가지를 재귀적으로 뻗어 만들기 때문에 매번 같은 모양이 나옵니다. 가지·잎·꽃은 모두 InstancedMesh로 그려서 수천 개를 띄워도 가볍게 돌아갑니다.
- **봄**: 연분홍 꽃이 가득 피고, 연둣빛 새잎이 돋으며, 꽃잎이 흩날립니다. 언덕에는 들꽃이 핍니다.
- **여름**: 짙은 초록 잎이 무성해지고 하늘이 짙은 파란색이 됩니다.
- **가을**: 잎마다 물드는 시점과 색(빨강·주황·노랑)이 달라 단풍이 차례로 듭니다. 낙엽은 바람에 흔들리며 떨어지고, 땅에 닿으면 잠시 쌓였다가 사라집니다.
- **겨울**: 잎이 다 떨어져 가지만 남고, 가지에는 서리 빛이 돕니다. 눈이 내리고 언덕은 하얗게 덮입니다.
- **표시**: 왼쪽 위에 지금 계절 이름(한글과 영문)과 진행 막대가 나옵니다. 계절이 바뀌면 이름이 페이드되며 바뀝니다.
- **외부 의존성**: three.js 0.160.0 하나만 jsDelivr에서 버전을 고정한 주소로 불러옵니다. 카메라 회전에 쓰는 OrbitControls도 같은 패키지에 들어 있는 애드온입니다. 이미지 파일은 쓰지 않고, 눈송이 텍스처는 캔버스로 직접 그려 만듭니다.
